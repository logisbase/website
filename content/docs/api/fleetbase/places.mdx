<Endpoint method="POST" path="/v1/places" title="Create a Place" id="create-a-place">
  Creates a place for the current company. Provide structured address fields, a free-form `address` or `street1` value to geocode, or coordinates for LogisBase to reverse geocode.

  <ParamTable title="Body parameters">
    <Param name="name" type="string">
      Display name for the place. Required on create unless address, street, or coordinates are enough for LogisBase to geocode the place.
    </Param>
    <Param name="address" type="string">
      Address string to geocode into a place when structured address fields are not supplied.
    </Param>
    <Param name="street1" type="string">
      Primary street address. Required on create unless address or coordinates are supplied.
    </Param>
    <Param name="street2" type="string">
      Secondary street address.
    </Param>
    <Param name="city" type="string">
      City or locality for the address.
    </Param>
    <Param name="province" type="string">
      State, province, or region for the address.
    </Param>
    <Param name="postal_code" type="string">
      Postal or ZIP code for the address.
    </Param>
    <Param name="neighborhood" type="string">
      Neighborhood or local area for the address.
    </Param>
    <Param name="district" type="string">
      District or administrative subdivision for the address.
    </Param>
    <Param name="building" type="string">
      Building name, number, or additional address detail.
    </Param>
    <Param name="security_access_code" type="string">
      Access code or delivery instructions needed to enter the building.
    </Param>
    <Param name="country" type="string">
      Country code for the address.
    </Param>
    <Param name="phone" type="string">
      Phone number associated with the place.
    </Param>
    <Param name="location" type="object">
      Resolvable point used as the place location. LogisBase accepts supported coordinate formats and reverse geocodes missing address fields.
    </Param>
    <Param name="latitude" type="number">
      Decimal latitude. Required with `longitude`.
    </Param>
    <Param name="longitude" type="number">
      Decimal longitude. Required with `latitude`.
    </Param>
    <Param name="owner" type="string">
      Contact or vendor ID that owns the place.
    </Param>
    <Param name="customer" type="string">
      Contact or vendor ID to assign as the place owner.
    </Param>
    <Param name="contact" type="string">
      Contact or vendor ID to assign as the place owner.
    </Param>
    <Param name="vendor" type="string">
      Vendor ID to assign as the place owner.
    </Param>
    <Param name="type" type="string">
      Place classification.
    </Param>
    <Param name="meta" type="object">
      Additional place metadata.
    </Param>
  </ParamTable>
