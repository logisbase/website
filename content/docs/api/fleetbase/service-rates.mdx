<Endpoint method="POST" path="/v1/service-rates" title="Create a Service Rate" id="create-a-service-rate">
  Create a Service Rate.

  <ParamTable title="Body parameters">
    <Param name="service_name" type="string" required>
      Display name for the service rate.
    </Param>
    <Param name="service_type" type="string" required>
      Service type this rate applies to.
    </Param>
    <Param name="service_area" type="string">
      Service area ID this rate applies to.
    </Param>
    <Param name="zone" type="string">
      Zone ID this rate applies to.
    </Param>
    <Param name="rate_calculation_method" type="enum" required>
      Method used to calculate the rate. One of `fixed_meter`, `fixed_rate`, `per_meter`, `per_drop`, `algo`, `parcel`.
    </Param>
    <Param name="currency" type="currency" required>
      Three-letter currency code.
    </Param>
    <Param name="base_fee" type="currency">
      Base fee applied to the rate.
    </Param>
    <Param name="per_meter_unit" type="enum">
      Distance unit for `per_meter` rates. One of `km`, `m`, `ft`, `yd`, `mi`.
    </Param>
    <Param name="per_meter_flat_rate_fee" type="currency">
      Flat fee per distance unit for `per_meter` rates.
    </Param>
    <Param name="meter_fees" type="array of objects">
      Distance fee tiers for `fixed_meter` or `fixed_rate` methods.
    </Param>
    <Param name="algorithm" type="string">
      Computable algorithm used when `rate_calculation_method` is `algo`.
    </Param>
    <Param name="has_cod_fee" type="boolean">
      Whether the rate includes a cash-on-delivery fee.
    </Param>
    <Param name="cod_calculation_method" type="enum">
      Cash-on-delivery fee calculation method. One of `percentage`, `flat`.
    </Param>
    <Param name="cod_flat_fee" type="currency">
      Flat cash-on-delivery fee.
    </Param>
    <Param name="cod_percent" type="integer">
      Percentage cash-on-delivery fee.
    </Param>
    <Param name="has_peak_hours_fee" type="boolean">
      Whether the rate includes a peak-hours fee.
    </Param>
    <Param name="peak_hours_calculation_method" type="enum">
      Peak-hours fee calculation method. One of `percentage`, `flat`.
    </Param>
    <Param name="peak_hours_flat_fee" type="currency">
      Flat peak-hours fee.
    </Param>
    <Param name="peak_hours_percent" type="integer">
      Percentage peak-hours fee.
    </Param>
    <Param name="peak_hours_start" type="string">
      Peak-hours start time in `HH:mm` format.
    </Param>
    <Param name="peak_hours_end" type="string">
      Peak-hours end time in `HH:mm` format.
    </Param>
    <Param name="duration_terms" type="string">
      Human-readable duration terms.
    </Param>
    <Param name="estimated_days" type="integer">
      Estimated service duration in days.
    </Param>
  </ParamTable>
