<Endpoint method="POST" path="/v1/zones" title="Create a Zone" id="create-a-zone">
  Creates a zone inside a service area. Provide either a GeoJSON boundary or a center point with radius, and LogisBase will store the zone for geofencing and service coverage checks.

  <ParamTable title="Body parameters">
    <Param name="name" type="string" required>
      Zone display name.
    </Param>
    <Param name="service_area" type="string" required>
      Service area ID the zone belongs to.
    </Param>
    <Param name="border" type="object">
      Zone polygon border. Required on create unless a point location or latitude/longitude is supplied.
    </Param>
    <Param name="location" type="object">
      Resolvable point used to create a circular zone.
    </Param>
    <Param name="latitude" type="number">
      Decimal latitude for a circular zone. Required with `longitude`.
    </Param>
    <Param name="longitude" type="number">
      Decimal longitude for a circular zone. Required with `latitude`.
    </Param>
    <Param name="radius" type="number" default="500">
      Radius in meters when creating a circular zone.
    </Param>
    <Param name="description" type="string">
      Zone description.
    </Param>
    <Param name="color" type="string">
      Fill color used when rendering the zone.
    </Param>
    <Param name="stroke_color" type="string">
      Stroke color used when rendering the zone.
    </Param>
    <Param name="status" type="enum">
      Zone status. One of `active`, `inactive`.
    </Param>
    <Param name="trigger_on_entry" type="boolean">
      Whether entering this zone should trigger geofence events.
    </Param>
    <Param name="trigger_on_exit" type="boolean">
      Whether exiting this zone should trigger geofence events.
    </Param>
    <Param name="dwell_threshold_minutes" type="integer">
      Minutes inside the zone before dwell behavior is triggered.
    </Param>
    <Param name="speed_limit_kmh" type="integer">
      Speed limit in kilometers per hour for geofence events.
    </Param>
  </ParamTable>
