<Endpoint method="POST" path="/v1/service-areas" title="Create a Service Area" id="create-a-service-area">
  A service area is created by simply providing a city, province or country in which LogisBase will reverse geocode into a service area.
  
  If the service area cannot be reverse geocoded, LogisBase will return an error.

  <ParamTable title="Body parameters">
    <Param name="name" type="string" required>
      Service area display name.
    </Param>
    <Param name="country" type="string" required>
      Country code or country name for the service area.
    </Param>
    <Param name="type" type="string">
      Service area type.
    </Param>
    <Param name="status" type="enum">
      Service area status. One of `active`, `inactive`.
    </Param>
    <Param name="border" type="object">
      Service area polygon border. Required on create unless a point location or latitude/longitude is supplied.
    </Param>
    <Param name="location" type="object">
      Resolvable point used to create a circular service area.
    </Param>
    <Param name="latitude" type="number">
      Decimal latitude for a circular service area. Required with `longitude`.
    </Param>
    <Param name="longitude" type="number">
      Decimal longitude for a circular service area. Required with `latitude`.
    </Param>
    <Param name="radius" type="number" default="500">
      Radius in meters when creating a circular service area.
    </Param>
    <Param name="parent" type="string">
      Parent service area ID.
    </Param>
    <Param name="color" type="string">
      Fill color used when rendering the service area.
    </Param>
    <Param name="stroke_color" type="string">
      Stroke color used when rendering the service area.
    </Param>
    <Param name="trigger_on_entry" type="boolean">
      Whether entering this service area should trigger geofence events.
    </Param>
    <Param name="trigger_on_exit" type="boolean">
      Whether exiting this service area should trigger geofence events.
    </Param>
    <Param name="dwell_threshold_minutes" type="integer">
      Minutes inside the service area before dwell behavior is triggered.
    </Param>
    <Param name="speed_limit_kmh" type="integer">
      Speed limit in kilometers per hour for geofence events.
    </Param>
  </ParamTable>
