<Endpoint method="POST" path="/v1/entities" title="Create an Entity" id="create-an-entity">
  Creates an entity such as a parcel, package, or item. Attach it to a payload and optionally set a destination or waypoint within that payload route.

  <ParamTable title="Body parameters">
    <Param name="name" type="string" required>
      Display name for the entity.
    </Param>
    <Param name="type" type="string" required>
      Entity type, such as `parcel`, `package`, or another application-defined type.
    </Param>
    <Param name="internal_id" type="string">
      Optional internal identifier for the entity.
    </Param>
    <Param name="description" type="string">
      Human-readable entity description.
    </Param>
    <Param name="payload" type="string">
      Payload ID to attach the entity to. Required when `destination` or `waypoint` is supplied.
    </Param>
    <Param name="destination" type="string">
      Place ID within the payload route to use as the entity destination.
    </Param>
    <Param name="waypoint" type="string">
      Alias for `destination`; resolves a place within the payload route.
    </Param>
    <Param name="customer" type="string">
      Contact or vendor ID to assign as the entity customer.
    </Param>
    <Param name="driver" type="string">
      Driver ID to associate with the entity.
    </Param>
    <Param name="email" type="string">
      Contact email associated with the entity.
    </Param>
    <Param name="weight" type="number">
      Entity weight.
    </Param>
    <Param name="weight_unit" type="enum">
      Unit for `weight`. Required when `weight` is supplied. One of `g`, `oz`, `lb`, `kg`.
    </Param>
    <Param name="length" type="number">
      Entity length.
    </Param>
    <Param name="width" type="number">
      Entity width.
    </Param>
    <Param name="height" type="number">
      Entity height.
    </Param>
    <Param name="dimensions_unit" type="enum">
      Unit for length, width, and height. Required when dimensions are supplied. One of `cm`, `in`, `ft`, `mm`, `m`, `yd`.
    </Param>
    <Param name="declared_value" type="currency">
      Declared value of the entity.
    </Param>
    <Param name="price" type="currency">
      Price of the entity.
    </Param>
    <Param name="sale_price" type="currency">
      Sale price of the entity.
    </Param>
    <Param name="sku" type="string">
      Stock keeping unit for the entity.
    </Param>
    <Param name="currency" type="currency">
      Three-letter currency code for monetary fields.
    </Param>
    <Param name="meta" type="object">
      Additional entity metadata.
    </Param>
  </ParamTable>
