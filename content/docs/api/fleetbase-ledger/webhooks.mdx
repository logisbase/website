<Endpoint method="POST" path="/webhooks" title="Handle Gateway Webhook" id="handle-gateway-webhook">
  Receives an inbound webhook from a payment gateway driver. LogisBase verifies the gateway event, records it idempotently, and dispatches normalized payment events.

  <ParamTable title="Body parameters">
    <Param name="company_uuid" type="string">
      Organization UUID used to resolve the active gateway when the gateway driver supports multi-company callbacks.
    </Param>
    <Param name="id" type="string">
      Gateway event or transaction identifier from the provider payload.
    </Param>
    <Param name="type" type="string">
      Gateway event type, such as a payment succeeded or payment failed event.
    </Param>
    <Param name="data" type="object">
      Provider-specific webhook data passed to the gateway driver for parsing.
    </Param>
  </ParamTable>
